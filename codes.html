Bubble Sort
void BubbleSort(int a[], int n)
{
int i, t, j;
for(i=0;i<n-1;i++)
{
for(j=0;j<n-1;j++)
{
if(a[j]>a[j+1])
{
t = a[j];
a[j] = a[j+1];
a[j+1] = t;
}
}
}
}
Quick Sort
void QuickSort(int a[], int s, int f)
{
if(s>=f) return;
int i=s-1, j, t;
for(j=s;j<f;j++)
{
if(a[j]<a[f])
{
t=a[++i];
a[i]=a[j];
a[j]=t;
}
}
t=a[f];
a[f]=a[++i];
a[i]=t;
QuickSort(a,s,i-1);
QuickSort(a,i+1,f);
}
